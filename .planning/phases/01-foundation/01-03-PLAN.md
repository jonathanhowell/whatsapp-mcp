    Export function `registerRefreshSnapshotTool` that takes `(server: McpServer, getState: () => { config: Config; handle: DatabaseHandle }, setState: (handle: DatabaseHandle) => void)` and calls:

    ```typescript
    server.registerTool(
      "refresh_snapshot",
      {
        description: "Create a fresh snapshot of the WhatsApp database. Use this to get the latest messages without restarting the server.",
        annotations: { readOnlyHint: true },  // per MCP-02
      },
      async () => {
        const { config, handle } = getState();
        if (handle.mode === "live") {
          return {
            content: [{ type: "text", text: "Server is in live mode — no snapshot to refresh. Database reads are already against the live file." }],
          };
        }
        try {
          const newHandle = await refreshSnapshot(config, handle);
          setState(newHandle);
          return {
            content: [{ type: "text", text: "Snapshot refreshed successfully. Database now reflects the latest WhatsApp messages." }],
          };
        } catch (err: any) {
          return {
            content: [{ type: "text", text: `Failed to refresh snapshot: ${err.message}` }],
            isError: true,
          };
        }
      }
    );
    ```

    Create src/server.ts — McpServer factory:

    Export function `createServer(): McpServer` that:

    Create src/index.ts — Entry point:

    ```typescript
    import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
    import { loadConfig } from "./config/loader.js";
    import { openDatabase, type DatabaseHandle } from "./db/connection.js";
    import { createServer } from "./server.js";
    import { registerRefreshSnapshotTool } from "./tools/refresh-snapshot.js";
    import type { Config } from "./config/loader.js";

    async function main(): Promise<void> {
      const config = loadConfig();
      process.stderr.write(`WhatsApp MCP: Mode=${config.mode}, DB=${config.dbPath}\n`);

      let handle = await openDatabase(config);

      const server = createServer();

      // State accessors for tools
      const getState = () => ({ config, handle });
      const setState = (newHandle: DatabaseHandle) => { handle = newHandle; };

      registerRefreshSnapshotTool(server, getState, setState);

      const transport = new StdioServerTransport();
      await server.connect(transport);
      process.stderr.write("WhatsApp MCP: Server running on stdio\n");
    }

    main().catch((err) => {
      process.stderr.write(`Fatal: ${err.message}\n`);
      process.exit(1);
    });
    ```

    - Use `registerTool()` not deprecated `server.tool()` overloads
    - All tools must have `annotations: { readOnlyHint: true }` (per MCP-02)
